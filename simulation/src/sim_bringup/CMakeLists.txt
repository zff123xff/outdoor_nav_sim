cmake_minimum_required(VERSION 3.16)
project(simulation_bringup)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(px4_gps_imu REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_library(px4_classic_sensors SHARED src/px4_classic_sensors.cpp)
target_include_directories(px4_classic_sensors PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(px4_classic_sensors px4_gps_imu::px4_ecl ${GAZEBO_LIBRARIES})
ament_target_dependencies(px4_classic_sensors gazebo_ros rclcpp sensor_msgs geometry_msgs nav_msgs)
install(TARGETS px4_classic_sensors LIBRARY DESTINATION lib)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ../../config/ DESTINATION share/${PROJECT_NAME}/config)
install(PROGRAMS scripts/visualization_tf.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/lifecycle_service_relay.py scripts/lifecycle_startup_watchdog.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME lifecycle_relay_faults COMMAND /usr/bin/python3 ${CMAKE_CURRENT_SOURCE_DIR}/test/lifecycle_relay_test.py)
  set_tests_properties(lifecycle_relay_faults PROPERTIES TIMEOUT 60 ENVIRONMENT "ROS_DOMAIN_ID=81")
  add_test(NAME launch_profiles COMMAND /usr/bin/python3 ${CMAKE_CURRENT_SOURCE_DIR}/test/launch_profiles_test.py ${CMAKE_CURRENT_SOURCE_DIR}/launch/go2_px4_sim.launch.py)
  add_test(NAME hil_contract COMMAND /usr/bin/python3 ${CMAKE_CURRENT_SOURCE_DIR}/test/hil_contract_test.py ${CMAKE_CURRENT_SOURCE_DIR}/launch/go2_px4_sim.launch.py)
endif()
ament_package()
